class Solution {
public:
    vector<int> findDuplicates(vector<int>& nums) {
        vector<int>hash(nums.size()+1,0);
        vector<int>res;
        for(int i=0;i<nums.size();i++)
        {
            hash[nums[i]]++;
            if(hash[nums[i]]==2)
            {
                  res.push_back(nums[i]);
            }
        }
        return res;
    }
};

class Solution {
    public List<Integer> findDuplicates(int[] nums) {
         ArrayList<Integer>arr = new ArrayList<>();
         for(int i=0;i<nums.length;i++)
         {
            int index=Math.abs(nums[i])-1;
            if(nums[index]<0)
            {
                arr.add(Math.abs(nums[i]));
            }
            else
            {
                nums[index]=-nums[index];
            }
         }
         return arr;
    }
}


EXPLAINATION:
====================================================================
PROBLEM: FIND ALL DUPLICATES IN AN ARRAY
====================================================================

APPROACH: The "Index Marking" (Negation) Technique
Time Complexity: O(N) | Auxiliary Space: O(1)

Since all numbers in the array are strictly in the range [1, N] 
(where N is the size of the array), we can use the values of the 
array itself as INDICES to mark which numbers we have visited. 

For every number 'x' we encounter:
1. Find its corresponding index: index = abs(x) - 1.
2. Look at the value located at nums[index].
3. If nums[index] is POSITIVE: We haven't seen 'x' before. 
   Mark it as seen by making nums[index] NEGATIVE.
4. If nums[index] is NEGATIVE: We HAVE seen 'x' before. 
   This means 'x' is a duplicate. Add 'x' to the result list.

class Solution {
    public List<Integer> findDuplicates(int[] nums) {
        ArrayList<Integer> arr = new ArrayList<>();
        
        for (int i = 0; i < nums.length; i++) {
            // Get the absolute value to handle already-negated numbers
            int index = Math.abs(nums[i]) - 1;
            
            // If the value at this index is negative, it's a duplicate
            if (nums[index] < 0) {
                arr.add(Math.abs(nums[i]));
            } else {
                // Mark the number as visited by negating it
                nums[index] = -nums[index];
            }
        }
        
        return arr;
    }
}
