class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int>res;
        for(int i=0;i<nums1.size();i++)
        {
            auto it=std::find(nums2.begin(),nums2.end(),nums1[i]);
            int idx=std::distance(nums2.begin(),it);
                bool flag=false;
                for(int j=idx+1;j<nums2.size();j++)
                {
                    if(nums2[idx]<nums2[j])
                    {
                        res.push_back(nums2[j]);
                        flag=true;
                        break;
                    }
                }
                if(!flag)res.push_back(-1);
        }
        return res;
    }
};
