JAVA Iterative Approach:
class Solution {
    public int search(int[] arr, int target) {
        int low=0;
        int high=arr.length-1;
        while(low<=high)
        {
            int mid=low+(high-low)/2;
            if(arr[mid]==target)return mid;
            if(arr[mid]>target)high=mid - 1;
            if(arr[mid]<target)low=mid+ 1;
        }
        return -1;
    }
}

C++ Recurrsive Approach:
class Solution {
public:
int binarySearch(vector<int>&nums,int key,int low,int high)
{
    if(low>high)return -1;
    int mid=low+(high-low)/2;
    if(nums[mid]==key)return mid;
    if(nums[mid]>key)return binarySearch(nums,key,low,mid- 1);
    if(nums[mid]<key)return binarySearch(nums,key,mid+1,high);
    return -1;
}
    int search(vector<int>& nums, int target) {
        return binarySearch(nums,target,0,nums.size()- 1);
    }
};
